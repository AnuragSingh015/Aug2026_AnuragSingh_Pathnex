🔹 Ansible — Setup Nginx with Custom Config


- name: Setup Nginx with Custom Config
hosts: all
became: yes
tasks: 
  - name: Install nginx
    yum:
       name: nginx
       state: present

  - name: Copy custom nginx.Config
    copy:
      src: nginx.Config
      dest: /etc/nginx/nginx.Conf
      
  - name: Start nginx service
    service:
       name: nginx
       state: started
       enabled: yes


# Terraform - Provision EC2 with IAM role

resource "aws_iam_role" "ec2_role {
  name              = "pathnex_ec2_role"

  assume_role_pollicy = jsonencode({
    Version = "2012-10-17"
    Statement = [
       {
        Action: "sts:AssumeRole"
        Effect = "Allow"
        Principal = {
            Service = "ec2.amazonaws.com"
        }
       },
    ]
  })
}

resource "aws_instance" "pathnex_ec2" {
    ami = "ami-0abc1abc2abc"
    instance_type = "t2.medium"
    iam_instance_profile = aws_iam_role.name
}
