# Ansible(Setup Auto-Scaling Group)
- name: Setup Auto Scaling Group
  hosts: all
  became: yes
  tasks: 
    - name: Create Launch Configuration
      ec2_asg: 
        name: pathnex-ec2_asg
        region: us-east-1
        launch_config_name: pathnex-launch-Configuration
        min_size: 1
        max_size: 3
        desired_capacity: 2
        vpc_zone_identifier: subnet-0a1b2c3

---
# Terraform(Create a Load Balancer with EC2 Instance)

resource "aws_lb" "pathnex_lb" {
name              = "pathnex_lb"
internal          = false
load_balancer_type = "application"
security_group = [aws_security_group.allow_ssh_http.id]
subnets           = [aws_subnet.public.id]
}


resource "aws_instance" "pathnex_ec2" {
ami = "ami-0abcd1234"
instance_type = "t2.micro"
security_groups = [aws_security_group.allow_ssh_http.name]
tags = {
    Name = "Pathnex_EC2"
    }
}

---

# Kubernetes(Horizontal Pod Autoscaler with Load Balancer)

apiVersion: apps/v1
kind: Deployment
metadata:
    name: pathnex-web-application
spec:
  replicas:2
  selector:
    matchLabels:
        app: pathnex-web-app
  template: 
    metadata:
    labeles:
        app: pathnex-web-app
    spec: 
        container: 
            - name: web
              image: nginx
              ports:
               - containerPort:80

---
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-web-app-hpa
spec:
    scaleTargetRef:
     apiVersion: apps/v1
     kind: Deployment
     name: pathnex-web-app
    minReplicas: 1
    maxReplicas: 5
    targetCPUUtilizationPercentage: 80
